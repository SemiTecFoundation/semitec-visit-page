blink.sv
module blink (
  input  logic clk,
  input  logic rst_n,
  output logic led
);
  logic [26:0] count;

  always_ff @(posedge clk or negedge rst_n)
    if (!rst_n) count <= '0;
    else        count <= count + 1'b1;

  assign led = count[26];
endmodule
